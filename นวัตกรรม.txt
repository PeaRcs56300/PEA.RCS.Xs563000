const getMonthLabel = (monthKey) => {
  if (!monthKey || typeof monthKey !== 'string') return '';
  const parts = monthKey.split('-');
  if (parts.length < 2) return monthKey;
  const year = parts[0];
  const monthIdx = parseInt(parts[1]) - 1;
  const monthName = THAI_MONTHS[monthIdx] || 'ไม่ระบุ';
  return `${monthName} ${year}`;
};

const renderValue = (val) => {
  if (val === null || val === undefined) return '';
  if (typeof val === 'string' || typeof val === 'number' || typeof val === 'boolean') return String(val);
  return JSON.stringify(val);
};

const App = () => {
  const [user, setUser] = useState(null);
  const [data, setData] = useState([]);
  const [fileName, setFileName] = useState('');
  
  // Alert Settings
  const [alertThreshold, setAlertThreshold] = useState(205); 
  
  // Filter States
  const [fromMonth, setFromMonth] = useState('');
  const [fromDay, setFromDay] = useState('');
  const [toMonth, setToMonth] = useState('');
  const [toDay, setToDay] = useState('');
  const [startTime, setStartTime] = useState('00:00');
  const [endTime, setEndTime] = useState('23:59');
  
  const [isAdmin, setIsAdmin] = useState(false);
  const [isSyncing, setIsSyncing] = useState(false);
  const [lastSynced, setLastSynced] = useState(null);
  const [showLoginModal, setShowLoginModal] = useState(false);
  const [passInput, setPassInput] = useState('');
  const [loginError, setLoginError] = useState(false);
  
  const [focusedRecord, setFocusedRecord] = useState(null);
  const tableRef = useRef(null);
  const reportRef = useRef(null); 
  const [isGeneratingPDF, setIsGeneratingPDF] = useState(false);
  const [isPdfMode, setIsPdfMode] = useState(false); 

  useEffect(() => {
    const initAuth = async () => {
      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }
      } catch (err) { console.error("Auth Error:", err); }
    };
    initAuth();
    const unsubscribe = onAuthStateChanged(auth, setUser);
    return () => unsubscribe();
  }, []);

  useEffect(() => {
    if (!user) return;
    const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'power_history', 'latest_state');
    const unsubscribe = onSnapshot(docRef, (snapshot) => {
      if (snapshot.exists()) {
        const cloudData = snapshot.data();
        if (!isSyncing) {
          setData(Array.isArray(cloudData.logs) ? cloudData.logs : []);
          setFileName(String(cloudData.fileName || 'Cloud Data'));
          if (cloudData.timestamp) {
            const date = cloudData.timestamp.toDate ? cloudData.timestamp.toDate() : new Date(cloudData.timestamp);
            setLastSynced(date);
          }
        }
      }
    }, (error) => console.error("Snapshot error:", error));
    return () => unsubscribe();
  }, [user, isSyncing]);

  const uniqueMonths = useMemo(() => {
    if (!data.length) return [];
    const months = data.map(item => String(item?.Timestamp || '').split(' ')[0].substring(0, 7));
    return [...new Set(months.filter(m => m.length === 7))].sort();
  }, [data]);

  const getDaysInMonth = (monthStr) => {
    if (!data.length || !monthStr) return [];
    const days = data
      .filter(item => String(item?.Timestamp || '').startsWith(monthStr))
      .map(item => String(item?.Timestamp || '').split(' ')[0].split('-')[2]);
    return [...new Set(days.filter(d => d))].sort();
  };

  const tableHeaders = useMemo(() => {
    if (data.length === 0) return [];
    const allKeys = Object.keys(data[0]);
    const priority = ['Timestamp', 'Voltage LL1', 'Voltage LL2', 'Voltage LL3', 'Voltage LN1', 'Voltage LN2', 'Voltage LN3', 'Current 1', 'Current 2', 'Current 3', 'Power1', 'Power2', 'Power3', 'Total Power'];
    const others = allKeys.filter(k => !priority.includes(k));
    return [...priority.filter(k => allKeys.includes(k)), ...others];
  }, [data]);

  useEffect(() => {
    if (uniqueMonths.length > 0 && !fromMonth) {
      const startMonth = uniqueMonths[0];
      const endMonth = uniqueMonths[uniqueMonths.length - 1];
      setFromMonth(startMonth);
      setToMonth(endMonth);
      
      const startDays = getDaysInMonth(startMonth);
      const endDays = getDaysInMonth(endMonth);
      if (startDays.length > 0) setFromDay(startDays[0]);
      if (endDays.length > 0) setToDay(endDays[endDays.length - 1]);
    }
  }, [uniqueMonths, fromMonth]);

  const filteredData = useMemo(() => {
    let result = data;
    const startStr = `${fromMonth}-${fromDay}`;
    const endStr = `${toMonth}-${toDay}`;

    if (fromMonth && fromDay && toMonth && toDay) {
      result = result.filter(item => {
        const datePart = String(item?.Timestamp || '').split(' ')[0];
        return datePart >= startStr && datePart <= endStr;
      });
    }

    return result.filter(item => {
      const timestamp = String(item?.Timestamp || '');
      const timePart = timestamp.includes(' ') ? timestamp.split(' ')[1] : "00:00";
      return timePart >= startTime && timePart <= endTime;
    });
  }, [data, fromMonth, fromDay, toMonth, toDay, startTime, endTime]);

  const latest = filteredData.length > 0 ? filteredData[filteredData.length - 1] : null;

  // --- COMPREHENSIVE ANALYSIS LOGIC ---
  const analysis = useMemo(() => {
    if (filteredData.length === 0) return { score: 0, avgV: '0.0', dipPercent: '0.00', avgUnbalance: '0.00', peakUnbalance: '0.00', avgDayPower: '0.00', minDayPower: '0.00', peakDayPower: '0.00', suggestedSolar: '0.0', savings: '0', dayEnergy: '0.0', nightEnergy: '0.0', dayRatio: '0' };

    const v1s = filteredData.map(d => parseFloat(d['Voltage LN1']) || 0).filter(v => v > 0);
    const v2s = filteredData.map(d => parseFloat(d['Voltage LN2']) || 0).filter(v => v > 0);
    const v3s = filteredData.map(d => parseFloat(d['Voltage LN3']) || 0).filter(v => v > 0);
    const allVs = [...v1s, ...v2s, ...v3s];
    const avgV = allVs.length > 0 ? allVs.reduce((a, b) => a + b, 0) / allVs.length : 0;
    
    const dipCount = filteredData.filter(item => {
        const v1 = parseFloat(item['Voltage LN1']) || 0;
        const v2 = parseFloat(item['Voltage LN2']) || 0;
        const v3 = parseFloat(item['Voltage LN3']) || 0;
        return (v1 > 0 && v1 < alertThreshold) || (v2 > 0 && v2 < alertThreshold) || (v3 > 0 && v3 < alertThreshold);
    }).length;

    const unbalanceList = filteredData.map(d => {
        const c1 = parseFloat(d['Current 1']) || 0;
        const c2 = parseFloat(d['Current 2']) || 0;
        const c3 = parseFloat(d['Current 3']) || 0;
        const avg = (c1 + c2 + c3) / 3;
        if (avg === 0) return 0;
        return (Math.max(Math.abs(c1 - avg), Math.abs(c2 - avg), Math.abs(c3 - avg)) / avg) * 100;
    });
    const avgUnbalance = unbalanceList.length > 0 ? unbalanceList.reduce((a, b) => a + b, 0) / unbalanceList.length : 0;
    const peakUnbalance = Math.max(...unbalanceList, 0);

    // Solar & Daytime Analysis
    const daytimeRecords = filteredData.filter(d => {
        const time = String(d?.Timestamp || '').split(' ')[1];
        return time >= "08:00" && time <= "17:00";
    });
    const nightRecords = filteredData.filter(d => {
        const time = String(d?.Timestamp || '').split(' ')[1];
        return time < "08:00" || time > "17:00";
    });

    const daytimePowers = daytimeRecords.map(d => parseFloat(d['Total Power']) || 0);
    const avgDayPower = daytimePowers.length > 0 ? daytimePowers.reduce((a, b) => a + b, 0) / daytimePowers.length : 0;
    const minDayPower = daytimePowers.length > 0 ? Math.min(...daytimePowers) : 0;
    const peakDayPower = daytimePowers.length > 0 ? Math.max(...daytimePowers) : 0;

    // Energy Calculation Estimation
    const totalEnergy = filteredData.length > 0 ? 
        (Math.max(...filteredData.map(d => parseFloat(d['EXP Energy']) || 0)) - Math.min(...filteredData.map(d => parseFloat(d['EXP Energy']) || 0))) : 0;

    const dayPowerSum = daytimeRecords.reduce((a, b) => a + (parseFloat(b['Total Power']) || 0), 0);
    const nightPowerSum = nightRecords.reduce((a, b) => a + (parseFloat(b['Total Power']) || 0), 0);
    const totalPowerSum = dayPowerSum + nightPowerSum;
    const dayEnergyEst = totalPowerSum > 0 ? (dayPowerSum / totalPowerSum) * totalEnergy : 0;
    const nightEnergyEst = totalEnergy - dayEnergyEst;
    const dayRatio = totalEnergy > 0 ? ((dayEnergyEst / totalEnergy) * 100).toFixed(0) : 0;

    const suggestedSolarSize = (avgDayPower * 0.8).toFixed(1); 
    const yearlySavingsEst = Math.round(parseFloat(suggestedSolarSize) * 4 * 365 * 4.42).toLocaleString();

    const vScore = Math.max(0, 100 - (dipCount / filteredData.length) * 500);
    const cScore = Math.max(0, 100 - avgUnbalance * 2);
    const finalScore = Math.round((vScore * 0.6) + (cScore * 0.4));

    return {
        score: finalScore,
        avgV: avgV.toFixed(1),
        dipPercent: ((dipCount / filteredData.length) * 100).toFixed(2),
        avgUnbalance: avgUnbalance.toFixed(2),
        peakUnbalance: peakUnbalance.toFixed(2),
        avgDayPower: avgDayPower.toFixed(2),
        minDayPower: minDayPower.toFixed(2),
        peakDayPower: peakDayPower.toFixed(2),
        suggestedSolar: suggestedSolarSize,
        savings: yearlySavingsEst,
        dayEnergy: dayEnergyEst.toFixed(1),
        nightEnergy: nightEnergyEst.toFixed(1),
        dayRatio: dayRatio
    };
  }, [filteredData, alertThreshold]);

  const currentUnbalance = useMemo(() => {
    if (!latest) return "0.00";
    const c1 = parseFloat(latest['Current 1']) || 0;
    const c2 = parseFloat(latest['Current 2']) || 0;
    const c3 = parseFloat(latest['Current 3']) || 0;
    const avg = (c1 + c2 + c3) / 3;
